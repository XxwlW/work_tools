const Utils = {
    ...CoreUtils,
    formatCurrency(amount) {
        const value = this.normalizeNumber(amount, 0);
        const sign = value < 0 ? '-' : '';
        return `${sign}￥${Math.abs(value).toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
    },
    formatDate(date) {
        return this.formatDateKey(date);
    },
    getWeekday(date) {
        const weekDays = ['周日', '周一', '周二', '周三', '周四', '周五', '周六'];
        return weekDays[date.getDay()];
    },
    stringToHash(str) {
        let hash = 0;
        for (let i = 0; i < str.length; i++) {
            hash = ((hash << 5) - hash) + str.charCodeAt(i);
            hash |= 0;
        }
        return hash;
    },
    getTagStyle(tagName) {
        const hash = this.stringToHash(tagName);
        const hue = Math.abs(hash % 360);
        const color = `hsl(${hue}, 60%, 35%)`;
        const bg = `hsl(${hue}, 80%, 92%)`;
        return `background-color: ${bg}; color: ${color}; padding: 0 4px; border-radius: 3px; font-size: 0.8em; font-weight: 600; border: 1px solid ${color}; display: inline-block; line-height: 1.2;`;
    },
    getTagColor(tagName, opacity = 0.8) {
        const hash = this.stringToHash(tagName);
        const hue = Math.abs(hash % 360);
        return `hsla(${hue}, 70%, 55%, ${opacity})`;
    }
};

// --- 3. 数据核心 (保持 v2.5 原样逻辑) ---
async function loadAndProcessTransactions(startDate, endDate) {
    console.log(`分析时段: ${startDate} to ${endDate}`);
    const walletLinks = CoreUtils.collectWalletPages({ dv }).map(page => [page.file.link || { path: page.file.path }]);
    if (!walletLinks || walletLinks.length === 0) {
        dv.paragraph("❌ **错误**: 未能找到任何钱包文件。"); return [];
    }

    const allTransactions = [];
    const startDateObj = Utils.resolveBillDate(startDate);
    const endDateObj = Utils.resolveBillDate(endDate);
    endDateObj.setHours(23, 59, 59, 999);

    const walletPaths = {}; // Map<Name, Path>
    const { Wallet } = core; // Use Wallet class from Core
    if (typeof Wallet.resetParseCache === "function") Wallet.resetParseCache();

    for (const linkArr of walletLinks) {
        const link = linkArr[0];
        // 1. 使用 Core 的 Wallet 类统一解析
        const wallet = new Wallet(link);
        if (!wallet.name) continue;

        walletPaths[wallet.name] = wallet.path;

        // 2. 遍历该钱包的所有交易
        for (const t of wallet.transactions) {
            // 2.1 日期过滤
            const tDate = Utils.resolveBillDate(t.time);
            if (!tDate) continue;
            if (tDate < startDateObj || tDate > endDateObj) continue;

            // 2.2 排除转账 (使用配置的标签)
            const tags = Utils.normalizeArrayField(t.tags).map(tag => tag.replace(/^#/, ""));
            if (tags.includes(CONFIG.transferTag)) continue;

            // 2.3 分类逻辑 (保持原有的分类偏好逻辑)
            // Core 返回的 tags 已经去除了 # 号
            let category = '未分类';
            const journalTagIndex = tags.indexOf(CONFIG.journalTag);
            // 优先取 #记账 后面的标签
            if (journalTagIndex !== -1 && journalTagIndex + 1 < tags.length) {
                category = tags[journalTagIndex + 1];
            } else {
                // 否则取第一个非功能性标签
                const otherTags = tags.filter(tag => tag !== CONFIG.journalTag && tag !== '贷款');
                if (otherTags.length > 0) category = otherTags[0];
            }

            // 2.4 映射到分析面板的数据格式
            const amount = Utils.normalizeNumber(t.value, 0);
            allTransactions.push({
                amount,
                abs_amount: Math.abs(amount),
                category: category,
                description: t.displayText || t.text,
                displayText: t.displayText || t.text,
                displayParts: t.displayParts || [],
                date: tDate,
                method: wallet.name,
                path: t.path,
                link: {
                    path: t.path,
                    blockId: t.blockId || null,
                    subpath: t.blockId ? null : (t.lineIndex ?? t.line ?? null)
                },
                walletPath: wallet.path,
                tags,
                timestamp: tDate.getTime()
            });
        }
    }
    return { allTransactions, walletPaths };
}

// --- 4. 数据分析 ---
function analyzeData(transactions, walletPaths, startDate, endDate) {
    if (!transactions || transactions.length === 0) return null;
    const incomeData = transactions.filter(t => t.amount > 0);
    const expenseData = transactions.filter(t => t.amount < 0);

    // 趋势图数据准备
    const dailyTrendLabels = [];
    const dailyExpenseData = [];
    const dailyExpenseDetails = [];
    const dailyIncomeData = [];
    const dailyIncomeDetails = [];
    const dailyNetData = [];      // 累计净值
    const dailyDayNetData = [];   // 当日净值

    const s = new Date(startDate);
    const e = new Date(endDate);
    let loopLimit = 0;
    let cumulativeNet = 0; // 累计净值

    for (let d = new Date(s); d <= e && loopLimit < 366; d.setDate(d.getDate() + 1)) {
        const dateStr = Utils.formatDate(d);
        dailyTrendLabels.push(dateStr);

        // 支出数据
        const dayExpenseTx = expenseData.filter(t => Utils.formatDate(t.date) === dateStr);
        dailyExpenseData.push(dayExpenseTx.reduce((sum, t) => sum + t.abs_amount, 0));
        dailyExpenseDetails.push(dayExpenseTx.sort((a, b) => b.abs_amount - a.abs_amount));

        // 收入数据
        const dayIncomeTx = incomeData.filter(t => Utils.formatDate(t.date) === dateStr);
        dailyIncomeData.push(dayIncomeTx.reduce((sum, t) => sum + t.amount, 0));
        dailyIncomeDetails.push(dayIncomeTx.sort((a, b) => b.amount - a.amount));

        // 净值计算
        const dayIncome = dayIncomeTx.reduce((sum, t) => sum + t.amount, 0);
        const dayExpense = dayExpenseTx.reduce((sum, t) => sum + t.amount, 0);
        const dayNet = dayIncome + dayExpense;  // 当日净值
        cumulativeNet += dayNet;
        dailyDayNetData.push(dayNet);
        dailyNetData.push(cumulativeNet);

        loopLimit++;
    }

    const analysis = {
        summary: {
            totalIncome: incomeData.reduce((sum, t) => sum + t.amount, 0),
            totalExpense: expenseData.reduce((sum, t) => sum + t.amount, 0),
        },
        trend: {
            labels: dailyTrendLabels,
            expense: {
                data: dailyExpenseData,
                details: dailyExpenseDetails
            },
            income: {
                data: dailyIncomeData,
                details: dailyIncomeDetails
            },
            net: {
                data: dailyNetData,
                dayData: dailyDayNetData
            }
        },
        expense: expenseData.length ? {
            by_category: Object.entries(expenseData.reduce((acc, t) => { acc[t.category] = (acc[t.category] || 0) + t.abs_amount; return acc; }, {})).sort((a,b) => b[1] - a[1]),
            by_method: Object.entries(expenseData.reduce((acc, t) => { acc[t.method] = (acc[t.method] || 0) + t.abs_amount; return acc; }, {})).sort((a,b) => b[1] - a[1]),
            top_transactions: [...expenseData].sort((a, b) => a.amount - b.amount).slice(0, CONFIG.topN_transactions)
        } : null,
        income: incomeData.length ? {
            by_category: Object.entries(incomeData.reduce((acc, t) => { acc[t.category] = (acc[t.category] || 0) + t.amount; return acc; }, {})).sort((a,b) => b[1] - a[1]),
            by_source: Object.entries(incomeData.reduce((acc, t) => { acc[t.method] = (acc[t.method] || 0) + t.amount; return acc; }, {})).sort((a,b) => b[1] - a[1]),
            top_transactions: [...incomeData].sort((a, b) => b.amount - a.amount).slice(0, CONFIG.topN_transactions)
        } : null
    };
    analysis.summary.netFlow = analysis.summary.totalIncome + analysis.summary.totalExpense;
    analysis.walletPaths = walletPaths;
    return analysis;
}

function transactionToFilterItem(t, index = 0) {
    const tags = [...new Set([t.category, ...(t.tags || [])]
        .map(tag => String(tag || "").replace(/^#/, "").trim())
        .filter(Boolean))];
    const links = [t.walletPath].filter(Boolean);
    return {
        id: `${t.path || t.description || "transaction"}:${t.timestamp || ""}:${index}`,
        name: t.description,
        text: t.description,
        displayText: `${t.description || ""} ${t.category || ""} ${t.method || ""}`,
        path: t.path,
        tags,
        links,
        date: t.date,
        ctime: t.date,
        money: t.amount,
        amount: t.amount,
        abs_amount: t.abs_amount,
        category: t.category,
        method: t.method,
        raw: t,
    };
}

const transactionSortFields = [
    { key: 'date', label: '日期', fn: (a, b) => ViewKit.toTimestamp(b.date) - ViewKit.toTimestamp(a.date) },
    { key: 'money', label: '金额', fn: (a, b) => Math.abs(b.money || 0) - Math.abs(a.money || 0) },
    { key: 'amount', label: '净额', fn: (a, b) => (b.amount || 0) - (a.amount || 0) },
    { key: 'category', label: '分类', fn: (a, b) => String(a.category || "").localeCompare(String(b.category || ""), "zh-CN") },
    { key: 'method', label: '账户', fn: (a, b) => String(a.method || "").localeCompare(String(b.method || ""), "zh-CN") },
];

async function main() {
    let startDate = CONFIG.startDate;
    let endDate = CONFIG.endDate;
    const today = new Date();

    // 日期逻辑优化：如果未指定，默认取过去30天
    if (!startDate && !endDate) {
        const pastDate = new Date(today);
        pastDate.setDate(today.getDate() - 30); // 往前推30天
        startDate = Utils.formatDate(pastDate);
        endDate = Utils.formatDate(today);
    } else if (startDate && !endDate) {
        endDate = Utils.formatDate(today);
    }

    try {
        const { allTransactions, walletPaths } = await loadAndProcessTransactions(startDate, endDate);
        const filterItems = allTransactions.map(transactionToFilterItem);
        dv.container.innerHTML = '';
        const filterHost = dv.container.createEl('div', { style: 'margin-bottom:1rem;' });
        const reportHost = dv.container.createEl('div');
        const renderVisibleTransactions = (visibleItems = []) => {
            const visibleTransactions = (visibleItems || []).map(item => item.raw);
            const analysisResults = analyzeData(visibleTransactions, walletPaths, startDate, endDate);
            createReport(analysisResults, startDate, endDate, { host: reportHost });
        };

        if (filterItems.length === 0) {
            filterHost.remove?.();
            renderVisibleTransactions([]);
        } else {
            new ViewKit.FilterBar(filterHost, {
                controls: ['search', 'tags', 'links', 'dateRange', 'sort'],
                sortFields: transactionSortFields,
                initial: { sort: 'date', sortAsc: false },
                storageKey: `${dv.current()?.file?.path || 'FinanceAnalysis'}:${startDate}:${endDate}`,
                emptyText: '无匹配流水，点击清除过滤器',
                onFilter: renderVisibleTransactions,
            }).bind(filterItems);
        }
    } catch (error) {
        console.error("脚本执行出错:", error);
        dv.paragraph(`❌ **错误**: ${error.message}`);
    }
}

main();
