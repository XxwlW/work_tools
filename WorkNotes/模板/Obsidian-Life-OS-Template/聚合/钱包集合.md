function createChart(initial, daily, days, bills) {
    const dailyMap = new Map(), detailMap = new Map();
    for (const b of bills) {
        if (!b.date) continue;
        const d = Utils.resolveBillDate(b.date);
        if (!d) continue;
        const k = d.setHours(0,0,0,0);
        dailyMap.set(k, (dailyMap.get(k)||0) + b.value);
        if (!detailMap.has(k)) detailMap.set(k, []);
        detailMap.get(k).push({ w: b.wallet, v: b.value, d: b.description });
    }

    const lbl = [], data = [];
    let cash = initial;
    const now = new Date(); now.setHours(0,0,0,0);

    for (let i = 0; i < days; i++) {
        const curr = new Date(now); curr.setDate(now.getDate() + i);
        if (i > 0) {
            const prev = new Date(curr); prev.setDate(curr.getDate() - 1);
            cash -= daily;
            if (dailyMap.has(prev.getTime())) cash += dailyMap.get(prev.getTime());
        }
        lbl.push(curr.toISOString().split('T')[0]);
        data.push(cash.toFixed(2));
    }

    // 渐变色配置
    const gradientColor = (ctx) => {
        const canvas = ctx.chart.ctx;
        const gradient = canvas.createLinearGradient(0, 0, 0, 400);
        gradient.addColorStop(0, 'rgba(75, 192, 192, 0.4)');
        gradient.addColorStop(1, 'rgba(75, 192, 192, 0.0)');
        return gradient;
    };

    return {
      type: 'line',
      data: {
          labels: lbl,
          datasets: [{
              label: '预测余额',
              data: data,
              borderColor: '#26a69a',
              backgroundColor: gradientColor,
              fill: true,
              tension: 0.4,
              pointRadius: 0,
              pointHitRadius: 15,
              borderWidth: 2
          }]
      },
      options: {
        responsive: true, maintainAspectRatio: false,
        interaction: { mode: 'index', intersect: false },
        plugins: {
          legend: { display: false },
          annotation: {
              annotations: {
                  line1: {
                      type: 'line', yMin: CONFIG.chart.safetyLine, yMax: CONFIG.chart.safetyLine,
                      borderColor: 'rgba(255, 99, 132, 0.5)', borderWidth: 1, borderDash: [6, 6],
                      label: { content: '安全水位', enabled: true, position: 'start' }
                  }
              }
          },
          tooltip: {
            backgroundColor: 'rgba(30, 30, 30, 0.9)',
            titleColor: '#fff',
            bodyColor: '#ccc',
            padding: 10,
            cornerRadius: 8,
            callbacks: {
              label: (ctx) => ` 💰 余额: ${Number(ctx.raw).toLocaleString()}`,
              footer: (items) => {
                const k = Utils.resolveBillDate(lbl[items[0].dataIndex])?.getTime();
                if (!k) return '';
                if (!detailMap.has(k)) return '';
                const grp = detailMap.get(k).reduce((a, x) => {
                    if (!a[x.w]) a[x.w] = { tx: [], tot: 0 };
                    a[x.w].tx.push(x); a[x.w].tot += x.v; return a;
                }, {});
                const lines = ['', '📅 当日变动:'];
                for (const w in grp) {
                    const color = grp[w].tot > 0 ? '🟢' : '🔴';
                    lines.push(`${color} ${w} (${grp[w].tot > 0 ? '+' : ''}${grp[w].tot.toFixed(0)})`);
                    grp[w].tx.sort((a, b) => Math.abs(b.v) - Math.abs(a.v)).slice(0,3).forEach(x => lines.push(`  • ${x.d || '无描述'}: ${x.v.toFixed(0)}`));
                }
                return lines.join('\n');
              }
            }
          }
        },
        scales: {
            y: {
                beginAtZero: false,
                grid: { color: 'rgba(255,255,255,0.05)' },
                ticks: { callback: v => v >= 1000 ? (v/1000).toFixed(1)+'k' : v }
            },
            x: { grid: { display: false }, ticks: { maxTicksLimit: 8 } }
        }
      }
    };
}

function lazyRenderChart(chartBox, chartFactory, fallbackText) {
    chartBox.empty?.();
    const placeholder = chartBox.createEl('div', {
        cls: 'text-muted',
        text: '图表将在进入可见区域后加载...',
        style: 'height:100%; display:flex; align-items:center; justify-content:center; border:1px dashed var(--background-modifier-border); border-radius:8px;'
    });
    let rendered = false;
    const render = () => {
        if (rendered) return;
        rendered = true;
        placeholder.remove?.();
        try {
            if (window.renderChart) window.renderChart(chartFactory(), chartBox);
            else chartBox.createEl('div', { cls: 'text-muted', text: fallbackText });
        } catch (error) {
            chartBox.createEl('div', { cls: 'text-muted', text: `现金流图表渲染失败：${error.message || error}` });
        }
    };

    if (!('IntersectionObserver' in window)) {
        render();
        return;
    }

    const observer = new IntersectionObserver(entries => {
        if (entries.some(entry => entry.isIntersecting)) {
            observer.disconnect();
            render();
        }
    }, { rootMargin: '120px 0px' });
    observer.observe(chartBox);
}

    });

    const upcoming = Array.from(groupedMap.values())
        .sort((a, b) => Utils.resolveBillDate(a.date) - Utils.resolveBillDate(b.date));

    if (upcoming.length === 0) {
        radarCard.createEl('div', { cls: 'text-muted', text: '近期无待办账单 🎉' });
    } else {
        const scrollBox = radarCard.createEl('div', { cls: 'scroll-container' });
        const tl = scrollBox.createEl('div', { cls: 'timeline' });
        const loadStatus = scrollBox.createEl('div', { cls: 'text-muted', style: 'font-size:0.8em; padding:6px 0; text-align:center;' });
        let renderedCount = 0;

        const renderBillGroup = (u) => {
            const date = Utils.resolveBillDate(u.date);
            const rel = Utils.getRelativeTimeDesc(u.date);
            const isIncome = u.total > 0;
            const count = u.items.length;

            // Tooltip 内容构造
            const tooltip = u.items.map(i => `${i.displayText || i.description || i.text || ""}: ${i.value}`).join('\n');
            const safeTooltip = Utils.escapeHtml(tooltip);
            const firstItem = u.items[0] || {};
            const safeWalletPath = Utils.escapeHtml(u.path || "");
            const safeWallet = Utils.escapeHtml(u.wallet || "");
            const firstDescriptionHtml = ViewKit.renderDisplayParts(firstItem.displayParts, {
                fallback: firstItem.displayText || firstItem.description || firstItem.text || ""
            });
            const sourceButton = ViewKit.renderSourceButton(firstItem, {
                className: "tl-source-link",
                label: "📄",
                style: "text-decoration:none; color:var(--c-text-muted); margin-right:4px;",
            });

            const item = tl.createEl('div', { cls: `timeline-item ${rel.type}` });
            item.setAttribute('title', tooltip); // 原生 Tooltip

            // 描述显示：如果是多笔，显示 "X笔账单"，否则显示具体描述 (带链接)
            const descHtml = count > 1
                ? `<span style="font-weight:bold; color:var(--c-text-main)">${count}笔账单</span>`
                : `<span class="tl-desc">${sourceButton}${firstDescriptionHtml}</span>`;

            item.innerHTML = `
                <div class="tl-date">
                    <div style="font-weight:bold">${rel.text}</div>
                    <div style="font-size:0.8em">${date ? `${date.getMonth()+1}/${date.getDate()}` : '-'}</div>
                </div>
                <div class="tl-content" title="${safeTooltip}">
                    <div style="display:flex; flex-direction:column;">
                        <div>${descHtml}</div>
                        <a class="internal-link" href="${safeWalletPath}" style="font-size:0.75em; color:var(--c-text-muted); text-decoration:none;">${Utils.getWalletIcon(u.wallet)} ${safeWallet}</a>
                    </div>
                    <div class="tl-amt ${isIncome ? 'text-success' : 'text-danger'}">
                        ${isIncome ? '+' : ''}${u.total.toFixed(0)}
                    </div>
                </div>
            `;
        };

        const renderNextBills = () => {
            const nextItems = upcoming.slice(renderedCount, renderedCount + CONFIG.billBatchSize);
            nextItems.forEach(renderBillGroup);
            renderedCount += nextItems.length;
            loadStatus.setText(renderedCount < upcoming.length
                ? `已显示 ${renderedCount}/${upcoming.length}，继续下滑加载`
                : `已显示全部 ${upcoming.length} 条`);
        };

        renderNextBills();
        scrollBox.addEventListener('scroll', () => {
            if (renderedCount >= upcoming.length) return;
            const nearBottom = scrollBox.scrollTop + scrollBox.clientHeight >= scrollBox.scrollHeight - 80;
            if (nearBottom) renderNextBills();
        });
    }

    // 右侧：钱包列表 (Fixed Height Scroll)
    const walletCol = gridRow.createEl('div', { cls: 'fin-col' });
    const walletCard = walletCol.createEl('div', { cls: 'fin-card' });
    walletCard.innerHTML = `<div class="fin-card-header"><div class="fin-title">👛 钱包详情</div></div>`;
    const walletFilterHost = walletCard.createEl('div', { cls: 'wallet-filter-host' });

    const scrollBox = walletCard.createEl('div', { cls: 'scroll-container' });
    const table = scrollBox.createEl('table', { cls: 'wallet-table' });
    table.innerHTML = `
        <thead>
            <tr>
                <th>钱包 <span style="font-size:0.8em;opacity:0.6">(${walletSummaries.length})</span></th>
                <th>综合</th>
                <th>正值</th>
                <th>负债</th>
                <th>信用</th>
                <th>标签</th>
            </tr>
        </thead>
        <tbody></tbody>`;
    const tbody = table.querySelector('tbody');
    const walletCardList = scrollBox.createEl('div', { cls: 'wallet-card-list' });

    const walletSortFields = [
        { key: 'netWorth', label: '净值', fn: (a, b) => b.netWorth - a.netWorth },
        { key: 'positiveBalance', label: '正值', fn: (a, b) => b.positiveBalance - a.positiveBalance },
        { key: 'debt', label: '负债', fn: (a, b) => Math.abs(b.debt) - Math.abs(a.debt) },
        { key: 'creditLimit', label: '信用', fn: (a, b) => b.creditLimit - a.creditLimit },
        { key: 'name', label: '名称', fn: (a, b) => String(a.name || "").localeCompare(String(b.name || ""), "zh-CN") },
    ];

    const renderWalletSummaries = (summaries) => {
        tbody.innerHTML = '';
        walletCardList.innerHTML = '';
        const headCount = table.querySelector('thead th span');
        if (headCount) headCount.textContent = `(${summaries.length})`;

        if (summaries.length === 0) {
            const row = tbody.insertRow();
            walletCardList.createEl('div', { cls: 'text-muted', text: 'No wallets found' });
            row.innerHTML = `<td colspan="6" class="text-muted" style="text-align:center">未找到匹配钱包</td>`;
            return;
        }

        summaries.forEach(summary => {
            const row = tbody.insertRow();
            try {
                const safePath = Utils.escapeHtml(summary.path);
                const safeName = Utils.escapeHtml(summary.name);
                const safeTags = (summary.tags || []).map(t => Utils.escapeHtml(t)).join(', ') || '-';
                const debugBadge = summary.errors?.length
                    ? `<span class="badge text-warning" title="${Utils.escapeHtml(summary.errors.join('\n'))}">debug ${summary.errors.length}</span>`
                    : "";
                row.innerHTML = `
                    <td>
                        <a class="internal-link" href="${safePath}" style="font-weight:500; color:var(--c-text-main); text-decoration:none;">${safeName}</a>
                        ${debugBadge}
                    </td>
                    <td class="num-cell" style="font-weight:bold">${Utils.fmtMoney(summary.netWorth)}</td>
                    <td class="num-cell">${Utils.fmtMoney(summary.positiveBalance)}</td>
                    <td class="num-cell">${Utils.fmtMoney(summary.debt)}</td>
                    <td class="num-cell">${Utils.fmtMoney(summary.creditLimit)}</td>
                    <td class="num-cell" style="font-size:0.85em; color:var(--c-text-muted)">${safeTags}</td>
                `;
                const card = walletCardList.createEl('div', { cls: 'wallet-summary-card' });
                card.innerHTML = `
                    <div class="wallet-summary-head">
                        <a class="internal-link wallet-summary-title" href="${safePath}">${safeName}</a>
                        <div class="wallet-summary-net ${summary.netWorth >= 0 ? 'text-success' : 'text-danger'}">${Utils.fmtMoney(summary.netWorth)}</div>
                    </div>
                    <div class="wallet-summary-grid">
                        <div class="wallet-summary-stat"><div class="wallet-summary-label">正值</div><div class="wallet-summary-value text-success">${Utils.fmtMoney(summary.positiveBalance)}</div></div>
                        <div class="wallet-summary-stat"><div class="wallet-summary-label">负值</div><div class="wallet-summary-value text-danger">${Utils.fmtMoney(summary.debt)}</div></div>
                        <div class="wallet-summary-stat"><div class="wallet-summary-label">信用</div><div class="wallet-summary-value">${Utils.fmtMoney(summary.creditLimit)}</div></div>
                    </div>
                    <div class="wallet-summary-tags">${safeTags} ${debugBadge}</div>
                `;
            } catch (error) {
                row.innerHTML = `<td colspan="6" class="text-warning">钱包行渲染失败：${error.message || error}</td>`;
                walletCardList.createEl('div', { cls: 'wallet-summary-card text-warning', text: `Wallet row failed: ${error.message || error}` });
            }
        });
    };

    new ViewKit.FilterBar(walletFilterHost, {
        controls: ['search', 'tags', 'sort'],
        sortFields: walletSortFields,
        initial: { sort: 'netWorth', sortAsc: false },
        storageKey: `${dv.current()?.file?.path || 'WalletCollection'}:wallet-detail`,
        emptyText: '无匹配钱包，点击清除过滤器',
        onFilter: (visibleItems) => renderWalletSummaries((visibleItems || []).map(item => item.raw)),
    }).bind(walletFilterItems);
}

main();
