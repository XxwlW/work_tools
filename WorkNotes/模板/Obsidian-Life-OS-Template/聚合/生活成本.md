// --- 4. 趋势计算 ---
// calcPeriodAvg(item, periodStart, periodEnd, useActual)
//
// 关键设计：在役项（无 @）两条线共用同一时间范围 [start, today]，只有 rate 不同。
//   这样超预期项的实际 rate(cost/elapsed) < 预期 rate(cost/days)，实际线才能低于预期线。
//
// 预期线 rate = cost/days   （计划单价，固定）
// 实际线 rate = cost/elapsed （实际单价，超预期时随使用天数增长而降低）
//
// 有退役计划（@/@@）：两条线共用 [start, retiredDate]，rate 各自计算：
//   提前退役时 cost/actualDays > cost/days → 实际线 > 预期线（每天更贵）
//   寿终正寝(@@)时 actualDays=days → 两线重合
const calcPeriodAvg = (item, periodStart, periodEnd, useActual) => {
    if (!item.start || isNaN(item.start.getTime())) return 0;
    const activeStart = new Date(item.start);
    activeStart.setDate(activeStart.getDate() + 1);

    let activeEnd, rate;
    const planDays = item.actualDays != null && item.actualDays > 0 ? item.actualDays : null;

    if (planDays) {
        // 有退役计划：无论是否已到退役日，都按声明服役范围计算历史区间
        activeEnd = item.retiredDate && !isNaN(item.retiredDate.getTime())
            ? new Date(item.retiredDate)
            : new Date(item.start);
        if (!item.retiredDate || isNaN(item.retiredDate.getTime())) {
            activeEnd.setDate(activeEnd.getDate() + planDays);
        }
        rate = useActual
            ? (item.cost / planDays)        // cost/actualDays
            : (item.cost / item.days);      // cost/days（预期单价）
    } else {
        // 无退役计划的在役项（含超预期）：双线共用 [start, today]
        activeEnd = today;
        if (!useActual) {
            rate = item.cost / item.days;   // 预期线：始终用预期单价
        } else {
            // 实际线：以 periodEnd 节点的 elapsed 判断是否超过预期
            // 在预期期间内 → cost/days（两线重合）
            // 超出预期后  → cost/elapsed_at_that_point（实际摊薄，低于预期）
            const refDate = periodEnd < today ? periodEnd : today;
            const elapsedAtEnd = Math.max(0, Math.floor(
                (refDate.getTime() - item.start.getTime()) / 864e5
            ));
            rate = elapsedAtEnd > item.days
                ? item.cost / elapsedAtEnd  // 超预期：按该时间点的实际天数摊薄
                : item.cost / item.days;    // 在预期内：与预期单价相同
        }
    }

    const overlapStart = Math.max(activeStart.getTime(), periodStart.getTime());
    const overlapEnd   = Math.min(activeEnd.getTime(), periodEnd.getTime());
    const overlapDays  = Math.max(0, Math.floor((overlapEnd - overlapStart) / 864e5) + 1);
    if (overlapDays <= 0) return 0;
    const periodDays = Math.floor((periodEnd.getTime() - periodStart.getTime()) / 864e5) + 1;
    return (overlapDays / periodDays) * rate;
};


const buildWeekly = () => {
    const data = [];
    for (let w = 11; w >= 0; w--) {
        const wEnd = new Date(today); wEnd.setDate(today.getDate() - w * 7);
        const wStart = new Date(wEnd); wStart.setDate(wEnd.getDate() - 6);
        const exp = items.reduce((s, i) => s + calcPeriodAvg(i, wStart, wEnd, false), 0);
        const act = items.reduce((s, i) => s + calcPeriodAvg(i, wStart, wEnd, true), 0);
        const mm = String(wEnd.getMonth()+1).padStart(2,'0'), dd = String(wEnd.getDate()).padStart(2,'0');
        data.push({ label: `${mm}-${dd}`, exp, act });
    }
    return data;
};
const buildMonthly = () => {
    const data = [];
    for (let m = 5; m >= 0; m--) {
        const d = new Date(today); d.setMonth(d.getMonth() - m);
        const mStart = new Date(d.getFullYear(), d.getMonth(), 1);
        const mEnd   = new Date(d.getFullYear(), d.getMonth()+1, 0);
        const exp = items.reduce((s, i) => s + calcPeriodAvg(i, mStart, mEnd, false), 0);
        const act = items.reduce((s, i) => s + calcPeriodAvg(i, mStart, mEnd, true), 0);
        data.push({ label: `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`, exp, act });
    }
    return data;
};
const buildYearly = () => {
    const data = [];
    for (let m = 11; m >= 0; m--) {
        const d = new Date(today); d.setMonth(d.getMonth() - m);

        const mEnd   = new Date(d.getFullYear(), d.getMonth()+1, 0);

        const act = items.reduce((s, i) => s + calcPeriodAvg(i, mStart, mEnd, true), 0);
        data.push({ label: `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`, exp, act });
    }
    return data;
};

            { key: 'progress',  label: '进度' },
            { key: 'remaining', label: '剩余' },
            { key: null,        label: '退役日' },
            { key: null,        label: '状态' },
        ];
        const tr = thead.createEl('tr');
        for (const col of headerCols) {
            const th = tr.createEl('th', { text: col.label });
            if (col.label === '总价') th.style.textAlign = 'right';
            if (col.key) {
                if (sortKey === col.key) th.className = sortAsc ? 'sort-asc' : 'sort-desc';
                th.onclick = () => {
                    if (sortKey === col.key) { sortAsc = !sortAsc; }
                    else { sortKey = col.key; sortAsc = false; }
                    if (filterBar) filterBar.setState({ sort: sortKey, sortAsc });
                    else renderTable();
                };
            }
        }

        // 表体
        tbody.innerHTML = '';
        cardList.innerHTML = '';
        for (const item of filtered) {
            const progressUsedDays = item.progressDays ?? item.elapsed;
            const progressBaseDays = item.progressBaseDays || item.days;
            const pct = (item.progress * 100).toFixed(0);
            const isOver = !item.hasRetirementPlan && !item.isRetired && item.progress > 1; // 无退役计划且超预期服役
            const barColor = item.isRetired
                ? 'var(--c-text-muted)'
                : (isOver ? 'var(--c-warning)' : (item.progress > 0.8 ? 'var(--c-warning)' : 'var(--c-success)'));
            const badgeHtml = item.isRetired
                ? `<span class="lc-badge lc-badge-retired">⚪ 退役</span>`
                : (isOver
                    ? `<span class="lc-badge lc-badge-over">🔶 超预期</span>`
                    : `<span class="lc-badge lc-badge-active">🟢 服役中</span>`);
            const dailyDisplay = !item.isActive && !item.isRetired
                ? `<span style="color:var(--c-text-muted);">¥0</span>`
                : (() => {
                    const actual = item.actualDailyCost ?? item.dailyCost;
                    const planned = item.dailyCost;
                    if (isOver) {
                        // 超预期：实际日均更低，标注对比
                        return `<span style="font-weight:600;color:var(--c-success);">¥${actual.toFixed(2)}</span>` +
                               `<div style="font-size:0.7em;color:var(--c-text-muted);">预期¥${planned.toFixed(2)}</div>`;
                    }
                    return item.isActive
                        ? `<span style="font-weight:600;color:var(--c-accent);">¥${actual.toFixed(2)}</span>`
                        : `<span style="color:var(--c-text-muted);">¥${actual.toFixed(2)}</span>`;
                })();
            const remainText = item.isRetired
                ? '—'
                : (item.remaining < 0
                    ? `<span style="color:var(--c-warning)">超 ${Math.abs(item.remaining)}天</span>`
                    : `${item.remaining}天`);
            const startDisplay = fmtDate(item.start).slice(5); // MM-DD
            const retiredDisplay = item.retiredDate ? fmtDate(item.retiredDate) : (item.isRetired && item.actualDays != null ? `服${item.actualDays}天` : '—');
            const safeCategory = Utils.escapeHtml(item.category);
            const nameHtml = ViewKit.renderDisplayParts(item.displayParts, {
                fallback: item.displayText || item.name || ""
            });
            const sourceButton = ViewKit.renderSourceButton(item, {
                className: "lc-source-link",
                label: "📄",
                style: "text-decoration:none; color:var(--c-text-muted); margin-right:4px;",
            });

            const row = tbody.insertRow();
            row.innerHTML = `
              <td><span style="font-weight:500;color:var(--c-text-main);">${sourceButton}${nameHtml}</span>
                <div style="font-size:0.72em;color:var(--c-text-muted);">${safeCategory}</div></td>
              <td>${dailyDisplay}</td>
              <td style="text-align:right;font-family:monospace;">¥${item.cost.toLocaleString()}</td>
              <td style="color:var(--c-text-muted);font-size:0.88em;">${startDisplay}</td>
              <td style="min-width:90px;">
                <div class="lc-progress-bg"><div class="lc-progress-fill" style="width:${Math.min(Number(pct),100)}%;background:${barColor};"></div></div>
                <div style="font-size:0.7em;color:var(--c-text-muted);margin-top:2px;">${isOver ? `超预期 ${pct}%` : `${pct}%`} (${progressUsedDays}/${progressBaseDays}天)</div>
              </td>
              <td style="font-family:monospace;font-size:0.9em;">${remainText}</td>
              <td style="font-size:0.85em;color:var(--c-text-muted);">${retiredDisplay}</td>
              <td>${badgeHtml}</td>
            `;
            const card = cardList.createEl('div', { cls: 'lc-mobile-card' });
            card.innerHTML = `
              <div class="lc-mobile-head">
                <div class="lc-mobile-title">${sourceButton}${nameHtml}</div>
                <div class="lc-mobile-daily">${dailyDisplay}</div>
              </div>
              <div class="lc-mobile-meta">
                <span>${safeCategory}</span>
                <span>总价 ¥${item.cost.toLocaleString()}</span>
                <span>开始 ${startDisplay}</span>
                <span>${retiredDisplay}</span>
                ${badgeHtml}
              </div>
              <div class="lc-mobile-progress">
                <div>
                  <div class="lc-progress-bg"><div class="lc-progress-fill" style="width:${Math.min(Number(pct),100)}%;background:${barColor};"></div></div>
                  <div style="font-size:0.72em;color:var(--c-text-muted);margin-top:3px;">${isOver ? `超预期 ${pct}%` : `${pct}%`} (${progressUsedDays}/${progressBaseDays}天)</div>
                </div>
                <div style="font-family:monospace;font-size:0.9em;">${remainText}</div>
              </div>
            `;
        }
        if (filtered.length === 0) {
            const emptyRow = tbody.insertRow();
            emptyRow.innerHTML = `<td colspan="8" style="text-align:center;color:var(--c-text-muted);padding:1.5rem;">暂无数据</td>`;
            cardList.createEl('div', { cls: 'lc-mobile-card', text: '暂无数据' });
        }

        // 更新 tab 角标
        tabGrp.querySelectorAll('.lc-tab').forEach((btn, i) => {
            const cnt = items
                .filter(item => filteredItemIds.has(itemFilterId.get(item)))
                .filter(tabs[i].filter)
                .length;
            btn.innerHTML = `${tabs[i].label} <span class="lc-tab-count">${cnt}</span>`;
        });
    };

    tabs.forEach(({ key }) => {
        const btn = tabGrp.createEl('button', { cls: `lc-tab${key === activeTab ? ' active' : ''}` });
        btn.onclick = () => {
            tabGrp.querySelectorAll('.lc-tab').forEach(b => b.classList.remove('active'));
            btn.classList.add('active');
            activeTab = key;
            renderTable();
        };
    });

    filterBar = new ViewKit.FilterBar(filterHost, {
        controls: ['search', 'tags', 'links', 'sort'],
        sortFields: filterSortFields,
        initial: { sort: sortKey, sortAsc },
        storageKey: `${dv.current()?.file?.path || 'LivingCost'}:detail`,
        emptyText: '无匹配成本项目，点击清除过滤器',
        onFilter: (visibleFilterItems, state) => {
            filteredItemIds = new Set((visibleFilterItems || []).map(item => item.id));
            sortKey = state.sort || sortKey;
            sortAsc = Boolean(state.sortAsc);
            renderTable();
        },
    });
    filterBar.bind(filterItems);
}
