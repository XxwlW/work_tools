# 2 Frame Splitter

splitter.py

```python
def split_frames(text):

    frames = []

    current = []

    for line in text.splitlines():

        current.append(line)

        if "Current frame end." in line:

            frames.append("\n".join(current))

            current = []

    return frames
```

因为你的日志已经明确：

```text
Current frame end.
```

是帧结束符。

## geometry_parser.py

```python
import math

from model.point import Point

def valid_pt(x,y):

    if math.isnan(x):
        return False

    if math.isnan(y):
        return False

    if abs(x) > 50000:
        return False

    if abs(y) > 50000:
        return False

    return True
```

---

```python
def parse_point_block(block):

    pts=[]

    for x,y in POINT_RE.findall(block):

        x=float(x)
        y=float(y)

        if not valid_pt(x,y):
            continue

        pts.append(
            Point(
                x/1000.0,
                y/1000.0
            )
        )

    return pts
```

# 4 Fusion解析

由于日志固定：

```text
BSegData
...
NSegData
...
FusSegData
...
```

所以直接做状态机。

---

```python
class FusionParser:

    def parse(self,text):

        stages=[]

        current=None

        lines=text.splitlines()

        i=0

        while i<len(lines):

            line=lines[i]

            if "Success==" in line:

                if current:
                    stages.append(current)

                current=FusionStage(
                    name=line.strip("=")
                )

            elif "BSegData" in line:

                block=self.collect_block(lines,i)

                current.bseg=parse_point_block(block)

            elif "NSegData" in line:

                block=self.collect_block(lines,i)

                current.nseg=parse_point_block(block)

            elif "FusSegData" in line:

                block=self.collect_block(lines,i)

                current.fusseg=parse_point_block(block)

            i+=1

        if current:
            stages.append(current)

        return stages
```

# 5 车辆绘制

vehicle_draw.py

```python
import numpy as np

def build_vehicle_polygon(
    x,
    y,
    yaw_deg,
    length=4.87,
    width=1.87
):

    yaw=np.deg2rad(yaw_deg)

    corners=np.array([

        [ length/2, width/2],
        [ length/2,-width/2],

        [-length/2,-width/2],
        [-length/2, width/2],

        [ length/2, width/2]

    ])

    R=np.array([

        [np.cos(yaw),-np.sin(yaw)],
        [np.sin(yaw), np.cos(yaw)]

    ])

    pts=corners@R.T

    pts[:,0]+=x
    pts[:,1]+=y

    return pts
```
